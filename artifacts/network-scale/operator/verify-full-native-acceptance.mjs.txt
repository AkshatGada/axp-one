// Read-only chain verification of all funded opening and closing identities.
// Never loads a wallet, constructs a payment transport, signs or broadcasts.
import assert from 'node:assert/strict';
import {readFileSync,writeFileSync,existsSync} from 'node:fs';
import {resolve,join} from 'node:path';
import {parseEnv} from 'node:util';
import {createHash} from 'node:crypto';
import {createProductDevnetRPC,network} from '../axp-one/packages/product/native.mjs';
import {loadNativeSDK} from '../axp-one/packages/payments/sdk-loader.mjs';
const outer=import.meta.dirname,sha=x=>createHash('sha256').update(x).digest('hex');
const input=resolve(process.argv[2]??join(outer,'network-full-acceptance.json'));
const output=resolve(process.argv[3]??join(outer,'network-full-verification.json'));
assert.ok(input.startsWith(outer+'/')&&output.startsWith(outer+'/'));assert.equal(existsSync(output),false,'do not overwrite chain proof');
const bytes=readFileSync(input),record=JSON.parse(bytes),config=parseEnv(readFileSync(join(outer,'network-scale.env'),'utf8'));
let rpcCalls=0;const nativeRPC=createProductDevnetRPC(config.AXP_NETWORK_DEVNET_RPC_URL);
const rpc=(method,params)=>{assert.ok(['getGenesisHash','getSignatureStatuses','getTransaction','getAccountInfo'].includes(method));rpcCalls++;return nativeRPC(method,params);};
assert.equal(record.source.builderSHA256,sha(readFileSync(join(outer,'build-full-native-acceptance.mjs'))),'bound builder policy required');assert.equal(record.schemaVersion,'axp.product-native-acceptance.v1');assert.equal(record.mode,'devnet');assert.equal(record.network,'solana-devnet');assert.equal(record.genesisHash,network.genesisHash);assert.equal(record.program,network.program);assert.equal(record.mint,network.mint);assert.equal(record.scope.channels,125);assert.equal(record.scope.transactions,250);assert.equal(record.scope.parentRunStatus,'partial');assert.equal(record.scope.financialCleanupComplete,true);assert.equal(record.scope.planned128StagePassed,false);assert.equal(record.recovery.unfundedExpiredOpenings.length,3);assert.equal(record.recovery.retiredSignedCloses.length,5);
assert.equal(await rpc('getGenesisHash'),network.genesisHash);
const sdk=await loadNativeSDK(),program=await rpc('getAccountInfo',[record.program,{encoding:'base64',commitment:'finalized'}]);assert.equal(program.value.executable,true);assert.equal(program.value.owner,'BPFLoaderUpgradeab1e11111111111111111111111');const pb=Buffer.from(program.value.data[0],'base64');assert.equal(sha(pb),record.source.deployment.programAccountHash);assert.equal(pb.readUInt32LE(0),2);
const pd=await rpc('getAccountInfo',[sdk.kit.getAddressDecoder().decode(pb.subarray(4,36)),{encoding:'base64',commitment:'finalized',minContextSlot:program.context.slot}]);assert.equal(pd.value.owner,program.value.owner);assert.equal(sha(Buffer.from(pd.value.data[0],'base64')),record.source.deployment.programDataHash);
const pairs=record.campaigns.flatMap(c=>c.payment.transactions.map(t=>({c,t})));assert.equal(pairs.length,250);assert.equal(new Set(pairs.map(({t})=>t.signature)).size,250);assert.equal(new Set(record.campaigns.map(c=>c.payment.protocolChannelId)).size,125);
const statuses=(await rpc('getSignatureStatuses',[pairs.map(({t})=>t.signature),{searchTransactionHistory:true}])).value;assert.equal(statuses.length,250);
const sum=(rows,owner)=>(rows??[]).filter(x=>x.mint===record.mint&&x.owner===owner).reduce((n,x)=>n+BigInt(x.uiTokenAmount.amount),0n);
let opens=0,closes=0,deposit=0n,payout=0n,refund=0n,fees=0n,newRent=0n,reclaimedRent=0n;const verified=[];
for(const [i,{c,t}] of pairs.entries()){
 assert.match(t.signature,/^[1-9A-HJ-NP-Za-km-z]{80,90}$/);assert.equal(t.status,'finalized');assert.equal(t.finality,'finalized');assert.ok(statuses[i]);assert.equal(statuses[i].err,null);assert.equal(statuses[i].confirmationStatus,'finalized');assert.equal(statuses[i].slot,t.slot);
 const tx=await rpc('getTransaction',[t.signature,{encoding:'jsonParsed',commitment:'finalized',maxSupportedTransactionVersion:0}]);assert.ok(tx?.meta);assert.equal(tx.meta.err,null);assert.equal(tx.slot,t.slot);assert.equal(tx.transaction.signatures[0],t.signature);assert.equal(String(tx.meta.fee),t.networkFeeLamports);
 const keys=tx.transaction.message.accountKeys.map(k=>k.pubkey??k),p=c.payment;assert.ok(keys.includes(record.program));assert.ok(keys.includes(p.protocolChannelId),'each transaction must bind its exact channel');assert.equal(p.payer,record.payerOwner);assert.equal(p.payee,record.publisherOwner);assert.equal(p.mint,record.mint);assert.equal(p.acceptedBaseUnits,p.authorizedBaseUnits);assert.equal(p.settledBaseUnits,p.authorizedBaseUnits);assert.equal(BigInt(p.settledBaseUnits)+BigInt(p.refundBaseUnits),50000n);
 const signerKeys=tx.transaction.message.accountKeys.filter(k=>k.signer).map(k=>k.pubkey);assert.ok(signerKeys.includes(record.payerOwner));if(t.operation==='close')assert.ok(signerKeys.includes(record.publisherOwner));
 const deltas={payer:(sum(tx.meta.postTokenBalances,record.payerOwner)-sum(tx.meta.preTokenBalances,record.payerOwner)).toString(),publisher:(sum(tx.meta.postTokenBalances,record.publisherOwner)-sum(tx.meta.preTokenBalances,record.publisherOwner)).toString(),treasury:(sum(tx.meta.postTokenBalances,network.treasuryOwner)-sum(tx.meta.preTokenBalances,network.treasuryOwner)).toString()};assert.deepEqual(deltas,t.tokenDeltas);
 assert.equal(keys.length,tx.meta.preBalances.length);assert.equal(keys.length,tx.meta.postBalances.length);
 const added=tx.meta.postBalances.reduce((n,x,k)=>n+(keys[k]!==record.payerOwner&&tx.meta.preBalances[k]===0?BigInt(x):0n),0n),reclaimed=tx.meta.preBalances.reduce((n,x,k)=>n+(keys[k]!==record.payerOwner&&tx.meta.postBalances[k]===0?BigInt(x):0n),0n);assert.equal(added.toString(),t.newRentLamports);assert.equal(reclaimed.toString(),t.reclaimedRentLamports);
 assert.equal(deltas.treasury,'0');if(t.operation==='open'){assert.equal(deltas.payer,'-50000');assert.equal(deltas.publisher,'0');opens++;deposit+=50000n;}else{assert.equal(t.operation,'close');assert.equal(deltas.publisher,p.authorizedBaseUnits);assert.equal(deltas.payer,p.refundBaseUnits);closes++;payout+=BigInt(deltas.publisher);refund+=BigInt(deltas.payer);}
 fees+=BigInt(tx.meta.fee);newRent+=added;reclaimedRent+=reclaimed;verified.push({channelId:p.channelId,protocolChannelId:p.protocolChannelId,operation:t.operation,signature:t.signature,slot:t.slot,networkFeeLamports:t.networkFeeLamports,tokenDeltas:deltas});
 if((i+1)%25===0)console.log(JSON.stringify({readOnly:true,finalizedTransactionsVerified:i+1,total:250}));
}
assert.equal(opens,125);assert.equal(closes,125);assert.equal(deposit,6250000n);assert.equal(payout,1236000n);assert.equal(refund,5014000n);assert.equal(payout+refund,deposit);
for(const [key,value] of Object.entries({depositBaseUnits:deposit,settledBaseUnits:payout,refundBaseUnits:refund}))assert.equal(record.accounting[key],value.toString());
let count=0;for(const c of record.campaigns){assert.equal(c.payment.vouchers.length,c.deliveryCount);let cumulative=0n;for(const [i,v] of c.payment.vouchers.entries()){assert.equal(v.status,'authorized');assert.equal(v.sequence,String(i+1));assert.equal(v.incrementBaseUnits,'1500');cumulative+=BigInt(v.incrementBaseUnits);assert.equal(v.cumulativeAmountBaseUnits,cumulative.toString());count++;}assert.equal(cumulative.toString(),c.payment.authorizedBaseUnits);}assert.equal(count,824);
const result={schemaVersion:'axp.network-scale-financial-chain-verification.v1',status:'verified',verifiedAt:new Date().toISOString(),network:'solana-devnet',genesisHash:network.genesisHash,source:{runId:record.source.runId,acceptanceSHA256:sha(bytes),executionSourceHash:record.source.executionSourceHash,verifierSHA256:sha(readFileSync(import.meta.filename)),operatorPolicySHA256:record.source.operatorPolicySHA256},scope:{fundedChannels:125,finalizedOpeningTransactions:125,finalizedClosingTransactions:125,finalizedTransactions:250,offChainVoucherUpdates:824,voucherCryptographicVerification:'all 824 verified offline by the bound acceptance builder; RPC checks final cumulative payouts',parentRunStatus:'partial',planned128StagePassed:false,unfundedExpiredOriginalOpenings:3,retiredExpiredSignedCloseAttempts:5},accounting:{depositBaseUnits:deposit.toString(),publisherPayoutBaseUnits:payout.toString(),payerRefundBaseUnits:refund.toString(),networkFeeLamports:fees.toString(),newRentLamports:newRent.toString(),reclaimedRentLamports:reclaimedRent.toString()},checks:{devnetGenesis:true,deploymentHashes:true,uniqueExactSignatures:true,finalizedStatusAndSuccessfulTransactions:true,channelProgramAndExactChannelBound:true,payerAndClosePublisherSigners:true,tokenDeltasFeesAndRent:true,depositPayoutRefundConserved:true},operations:{providerCalls:0,rpcCalls,cryptographicSignaturesCreated:0,nativeBroadcasts:0,paymentStateWrites:0},transactions:verified,limitations:['Verification covers all 125 funded channels; the original planned 128-advertiser / 1,024-turn stage remains partial.','Three original expired/unfunded openings remain failed and retain reservations; expired close identities are preserved in the bound acceptance record.','One configured dedicated RPC provider; this is not independent-provider consensus.','Public Devnet test USDC and one shared sponsor; off-chain vouchers are cumulative authorizations, not 824 separate native transfers.']};
writeFileSync(output,JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({status:'verified',channels:125,transactions:250,vouchers:824,deposit:'6250000',publisherPayout:'1236000',payerRefund:'5014000',parentRunStatus:'partial',output,sha256:sha(readFileSync(output))}));
