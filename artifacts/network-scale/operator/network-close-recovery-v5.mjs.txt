function validateExpiryProof(ep,s,op,{fresh=true}={}){
 assert.equal(ep.schemaVersion,'axp.operator-expired-close-proof.v1');assert.equal(ep.outcome,'operator_proven_expired_unlanded');
 assert.equal(ep.sameProvider,true);assert.equal(ep.independentProviders,false);assert.equal(ep.signed,false);assert.equal(ep.broadcast,false);
 assert.equal(ep.channelId,s.channelId);assert.equal(ep.protocolChannelId,s.protocolChannelId);assert.equal(ep.closeId,op.id);
 assert.equal(op.signedRecordHash,hashNetworkRecord(op.signed));assert.equal(ep.signedRecordHash,op.signedRecordHash);
 assert.equal(ep.signature,op.signed.txSignature);assert.equal(op.attempt.txSignature,ep.signature);assert.equal(op.attempt.signedRecordHash,ep.signedRecordHash);
 assert.equal(ep.wireHash,sha(Buffer.from(op.signed.wireBase64,'base64')));assert.equal(ep.observations.length,2);
 const [a,b]=ep.observations;
 for(const o of [a,b]){
  assert.equal(o.signatureAbsent,true);assert.equal(o.transactionAbsent,true);assert.equal(o.blockhashValid,false);
  assert.equal(o.protocolChannelStatus,'Open');assert.equal(o.unspentDepositBaseUnits,s.terms.depositBaseUnits);assert.equal(o.deploymentVerified,true);
  assert.equal(String(o.lastValidHeight),String(op.signed.lastValidBlockHeight));assert.ok(BigInt(o.height)>BigInt(op.signed.lastValidBlockHeight));
  assert.ok(Number.isSafeInteger(o.slot)&&o.slot>0);assert.ok(Number.isSafeInteger(o.escrowSlot)&&o.escrowSlot>=o.slot);
  assert.match(o.channelAccountHash,/^[a-f0-9]{64}$/);assert.ok(Number.isFinite(Date.parse(o.at)));
 }
 assert.ok(Date.parse(b.at)-Date.parse(a.at)>=15000);assert.ok(b.slot>a.slot);assert.ok(b.escrowSlot>a.escrowSlot);assert.ok(b.height>a.height);
 assert.equal(a.channelAccountHash,b.channelAccountHash);
 if(fresh){const age=Date.now()-Date.parse(b.at);assert.ok(age>=0&&age<15*60*1000,'refresh readonly original-close proof before first archival');}
}

// The journal marker and conservative reservation commit in ONE transaction.
// A crash before the separate payment-store flag cannot add this fee twice.
function reservePlanFee(c,w,s,estimate){
 const nativeId=`${w.id}:${s.channelId}`,retired=s.retiredCloseAttempts??[];
 const markerId=hash({runId:s.terms.runId,workerId:w.id,channelId:s.channelId,closeId:s.close.id,termsHash:s.termsHash,watermarkHash:hash(s.watermark),retiredSignatures:retired.map(x=>x.operation.signed.txSignature)});
 return c.journal.tx(()=>{
  const reservation=c.journal.get('native',nativeId);assert.ok(reservation);
  const prior=c.journal.get('operatorFeeAdmission',markerId);
  if(prior){assert.ok(trustedRecoveryPolicy(prior.policyHash),'fee admission belongs to a pinned recovery policy');assert.equal(prior.nativeId,nativeId);assert.equal(prior.id,markerId);assert.deepEqual(prior.retiredSignatures,retired.map(x=>x.operation.signed.txSignature));}
  const baseline=money(prior?.baselineReservation??reservation.feeReserveLamports);
  const additional=retired.length?(estimate>money(prior?.additionalMaximum??'0')?estimate:money(prior?.additionalMaximum??'0')):0n;
  const ordinary=(money(s.open.signed.estimatedFeeAndRentLamports)>5000000n?money(s.open.signed.estimatedFeeAndRentLamports):5000000n)+(estimate>5000000n?estimate:5000000n);
  const target=retired.length?baseline+additional:ordinary;
  const next=target>money(reservation.feeReserveLamports)?target:money(reservation.feeReserveLamports);
  const others=c.journal.all('native').filter(x=>x.id!==nativeId);
  assert.ok(others.reduce((n,x)=>n+money(x.feeReserveLamports),0n)+next<=money(P.aggregateFeeRentLamports));
  assert.ok(others.filter(x=>x.id.startsWith(`${w.id}:`)).reduce((n,x)=>n+money(x.feeReserveLamports),0n)+next<=money(NATIVE_LIMITS.aggregateFeeRentLamports));
  if(next>money(reservation.feeReserveLamports))c.journal.put('native',nativeId,{...reservation,feeReserveLamports:next.toString()});
  const row={id:markerId,nativeId,policyHash:prior?.policyHash??policyHash,baselineReservation:baseline.toString(),additionalMaximum:additional.toString(),reservedTotal:next.toString(),retiredSignatures:retired.map(x=>x.operation.signed.txSignature)};
  if(!prior||hash(prior)!==hash(row)){c.journal.put('operatorFeeAdmission',markerId,row);c.journal.event('operator_close_fee_admitted',{...row,executingPolicyHash:policyHash,channelId:s.channelId,newDeposit:false});}
  return markerId;
 });
}

async function reconcileSaved(c,w,s){

 if(!ready){assert.ok(s.close.attempt);assert.equal(s.close.attempt.txSignature,signature);assert.equal(s.close.attempt.signedRecordHash,s.close.signedRecordHash);assert.ok(['submitted','submitting','unknown'].includes(s.close.status));}

 let r=await w.service.campaignAction(campaign.id,ready?'settle':'reconcile');
 for(let retry=0;r.campaign.payment.closeStatus!=='finalized'&&retry<8;retry++){
  assert.notEqual(r.campaign.payment.closeStatus,'failed','a failed saved close needs a separate explicit policy');
  await new Promise(resolve=>setTimeout(resolve,3000));r=await w.service.campaignAction(campaign.id,'reconcile');
 }
 const after=w.service.paymentWorker.store.get(s.channelId);
 assert.equal(after.close.signed.txSignature,signature);assert.equal(sha(Buffer.from(after.close.signed.wireBase64,'base64')),wireHash);

 c.journal.event('operator_saved_close_reconciled',{policyHash,workerId:w.id,channelId:s.channelId,signature});
}

    c.journal.tx(()=>{c.journal.put('operatorProof',hash(ep),ep);c.journal.event('operator_expired_signed_close_archived',{policyHash,proofPolicyHash:spec.producerPolicyHash,workerId:ew.id,channelId:es.channelId,signature:ep.signature,proofHash:hash(ep),sameProvider:true,newDeposit:false});});
    ew.service.paymentWorker.store.update(es.channelId,current=>{
     assert.equal(hash(current),hash(es));const archived={operation:structuredClone(current.close),proofHash:hash(ep),proofPolicyHash:spec.producerPolicyHash,recoveryPolicyHash:policyHash,outcome:ep.outcome,at:c.journal.stamp()};
     current.retiredCloseAttempts=[...(current.retiredCloseAttempts??[]),archived];
     current.close={id:current.close.id,status:'prepared',plan:current.close.plan,signed:null,receipt:null,operatorRecovery:{policyHash,proofHash:hash(ep),replacementAdmission:false}};return current;
    });
   }
  }
  // Restore the complete proven unsigned set before any signer can meet its barrier.
  for(const {w,s,proof} of unsigned){
   c.journal.tx(()=>{c.journal.put('operatorProof',hash(proof),proof);c.journal.event('operator_unsigned_close_restored',proof);});
   w.service.paymentWorker.store.update(s.channelId,current=>{assert.equal(hash(current),hash(s));current.retiredUnsignedClosePreparations=[...(current.retiredUnsignedClosePreparations??[]),{operation:structuredClone(current.close),proofHash:hash(proof)}];current.close.status='prepared';current.close.reasonCode=null;return current;});
  }
  // Reruns first resolve every previously saved replacement wire, never replace it.
  for(const w of c.workers)for(const s of w.service.paymentWorker.store.list()){
   if(s.open.status!=='finalized'){provenFailedOpen(c,s);continue;}
   if(s.close?.status==='finalized'){finalizedObligation(w,s);continue;}
   if(s.close?.signed)await reconcileSaved(c,w,s);
   else{frozenObligation(w,s);assert.equal(s.close.status,'prepared');assert.ok(!s.close.attempt);}
  }
  let closed=0;
  for(const w of c.workers)for(const original of w.service.paymentWorker.store.list()){
   let s=w.service.paymentWorker.store.get(original.channelId);if(s.open.status!=='finalized'||s.close?.status==='finalized')continue;
   frozenObligation(w,s);assert.equal(s.close.status,'prepared');assert.ok(!s.close.signed&&!s.close.attempt);
   const native=await createNativeTransport({terms:s.terms,statePath:join(w.stateDir,'native.sqlite'),walletPath:join(root,'local-state/product/secrets/devnet-wallet.json'),rpcURL:config.AXP_NETWORK_DEVNET_RPC_URL});
   let plan;try{plan=await native.prepareClose({channel:{protocolChannelId:s.protocolChannelId,channelId:s.channelId},...(s.intents.length?{finalVoucher:s.intents.at(-1).voucher}:{}),watermark:s.watermark});await native.assertPlanFresh({plan});}finally{native.close();}
   assert.equal(plan.protocolChannelId,s.protocolChannelId);assert.equal(hash(plan.watermark),hash(s.watermark));
   assert.equal(hash(plan.finalVoucher??null),hash(s.intents.at(-1)?.voucher??null));
   const estimate=money(plan.estimatedFeeAndRentLamports);assertRetainedCaps(c,w,s,estimate);
   const admissionMarker=reservePlanFee(c,w,s,estimate);assertRetainedCaps(c,w,s,estimate);
   w.service.paymentWorker.store.update(s.channelId,current=>{assert.equal(hash(current),hash(s));current.retiredUnsignedPlans=[...(current.retiredUnsignedPlans??[]),{plan:current.close.plan,planHash:hash(current.close.plan),policyHash}];current.close.plan=plan;if(current.close.operatorRecovery){current.close.operatorRecovery.replacementAdmission=true;current.close.operatorRecovery.feeAdmissionId=admissionMarker;}return current;});
   c.journal.event('operator_same_obligation_unsigned_plan_refreshed',{policyHash,channelId:s.channelId,closeId:s.close.id,watermarkHash:hash(s.watermark),feeAdmissionId:admissionMarker,replacementSignedAttempt:Boolean(s.retiredCloseAttempts?.length),newDeposit:false});
   const campaign=campaignFor(w,s);let r=await w.service.campaignAction(campaign.id,'settle');
   for(let retry=0;r.campaign.payment.closeStatus!=='finalized'&&retry<8;retry++){assert.notEqual(r.campaign.payment.closeStatus,'failed');await new Promise(resolve=>setTimeout(resolve,3000));r=await w.service.campaignAction(campaign.id,'reconcile');}
   assert.equal(r.campaign.payment.closeStatus,'finalized','resolve this exact saved identity before next signer');
   const finalized=w.service.paymentWorker.store.get(s.channelId);finalizedObligation(w,finalized);
   closed++;c.journal.event('operator_recovered_close_finalized',{policyHash,workerId:w.id,channelId:s.channelId,settledBaseUnits:finalized.close.receipt.settledBaseUnits,refundBaseUnits:finalized.close.receipt.refundBaseUnits,signature:finalized.close.signed.txSignature});
   console.log(JSON.stringify({workerId:w.id,channelId:s.channelId,status:'finalized',recoveredClosed:closed}));
  }
  const entries=c.workers.flatMap(w=>w.service.paymentWorker.store.list().map(s=>({w,s}))),reservations=c.journal.all('native');
  assert.deepEqual(entries.map(({w,s})=>`${w.id}:${s.channelId}`).sort(),reservations.map(x=>x.id).sort(),'every original native admission remains represented');
  let funded=0,unfunded=0;
  for(const {w,s} of entries){if(s.open.status==='finalized'){finalizedObligation(w,s);funded++;}else{provenFailedOpen(c,s);const totals=w.service.exchange.totals({channelId:s.channelId});assert.equal(totals.accepted,0n);assert.equal(totals.reserved,0n);unfunded++;}}
  assert.ok(funded>0);assert.equal(funded+unfunded,entries.length);
  assert.ok(reservations.reduce((n,x)=>n+money(x.feeReserveLamports),0n)<=money(P.aggregateFeeRentLamports));
  for(const w of c.workers)assert.ok(reservations.filter(x=>x.id.startsWith(`${w.id}:`)).reduce((n,x)=>n+money(x.feeReserveLamports),0n)<=money(NATIVE_LIMITS.aggregateFeeRentLamports));
  c.journal.update({status:'partial',settlementBlocked:unfunded,completedAt:c.journal.stamp()});
  console.log(JSON.stringify({status:'financial_cleanup_complete',additionalClosed:closed,fundedFinalizedChannels:funded,unfundedExpiredOpenings:unfunded,originalBenchmarkComplete:false}));
 }finally{c.close();}
